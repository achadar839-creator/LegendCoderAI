@app.route('/generate', methods=['POST'])
def generate():
    data = request.json or {}
    prompt = data.get('prompt', '')
    filename = os.path.basename(data.get('filename', 'output.py'))
    save_path = os.path.join("outputs", filename)

    ai_response = ai_engine.query_ai(prompt)
    if "Ollama Connection Error" in ai_response:
        code_content = f"# [Offline Fallback Mode - Ollama Not Running]\n# Prompt: {prompt}\n\ndef main():\n    print('Hello Chandan! Start Ollama in Termux for live DeepSeek AI generation.')\n\nif __name__ == '__main__':\n    main()\n"
        status_note = f"⚠️ नोट: {ai_response}\n(अभी बेसिक टेम्पलेट जनरेट किया गया है)"
    else:
        code_content = ai_response
        status_note = "✅ DeepSeek AI द्वारा कोड जनरेट किया गया!"

    try:
        with open(save_path, 'w', encoding='utf-8') as f:
            f.write(code_content)
        file_msg = f"फाइल 'outputs/{filename}' में सेव हो गई है!"
    except Exception as e:
        file_msg = f"फाइल सेव एरर: {str(e)}"

    return jsonify({"reply": f"{status_note}\n📁 {file_msg}\n\n{code_content}"})

# Voice-to-Code & Chat Engine: Cosmic Coder HD Voice with Conversational Support
import speech_recognition as sr
import pyttsx3

def init_tts():
    """
    HD और रियलिस्टिक स्पीच इंजन को इनिशियलाइज करता है।
    """
    engine = pyttsx3.init()
    engine.setProperty('rate', 175)
    
    voices = engine.getProperty('voices')
    for voice in voices:
        if "en" in voice.id.lower() or "zira" in voice.id.lower() or "guy" in voice.id.lower():
            engine.setProperty('voice', voice.id)
            break
            
    return engine

def speak(text):
    """
    कॉस्मिक कोडर की अपनी एचडी आवाज़ में जवाब देना (चाहे कोडिंग हो या नॉर्मल बातचीत)।
    """
    try:
        engine = init_tts()
        engine.say(text)
        engine.runAndWait()
    except Exception as e:
        print(f"[TTS Error]: {str(e)}")

def listen_and_converse(mode="code"):
    """
    यूजर की आवाज़ सुनता है। 
    अगर mode='chat' है, तो यह नॉर्मल बातचीत (Conversational AI) की तरह ट्रीट करेगा।
    अगर mode='code' है, तो यह कोडिंग प्रॉम्प्ट निकालेगा।
    """
    recognizer = sr.Recognizer()
    
    if mode == "code":
        welcome_msg = "Cosmic Core online. What shall we build in the universe today?"
    else:
        welcome_msg = "Cosmic Core listening. Tell me, what's on your mind?"
        
    print(f"🌌 {welcome_msg}")
    speak(welcome_msg)
    
    try:
        with sr.Microphone() as source:
            print("🎙️ Listening...")
            recognizer.adjust_for_ambient_noise(source, duration=1)
            audio = recognizer.listen(source, timeout=5, phrase_time_limit=15)
            
        print("⚡ Processing your voice...")
        text = recognizer.recognize_google(audio, language="hi-IN")
        
        return {"status": "success", "input_text": text}
        
    except sr.UnknownValueError:
        err_msg = "I didn't catch that. Could you say it again?"
        speak(err_msg)
        return {"status": "error", "message": err_msg}
    except sr.RequestError:
        err_msg = "Cosmic satellite connection lost."
        speak(err_msg)
        return {"status": "error", "message": err_msg}
    except Exception as e:
        return {"status": "critical_error", "message": str(e)}
